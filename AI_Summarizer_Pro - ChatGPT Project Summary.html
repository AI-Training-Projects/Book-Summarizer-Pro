from flask import Flask, render_template, request, send_file, redirect, url_for, flash, Response
from werkzeug.utils import secure_filename
from summarize import summarize_document
from config_manager import load_config, save_config
from utils import extract_text_from_pdf, extract_text_from_html, save_summary_files, LOG_FILE, get_logs
from functools import wraps

app = Flask(__name__)
app.secret_key = 'supersecretkey'
UPLOAD_FOLDER = "uploads"
SUMMARY_FOLDER = "summaries"
CONFIG_PATH = "config/settings.json"
os.makedirs(UPLOAD_FOLDER, exist_ok=True)
os.makedirs(SUMMARY_FOLDER, exist_ok=True)
os.makedirs("logs", exist_ok=True)

# Admin credentials
USERNAME = 'admin'
PASSWORD = 'Admin123'

def check_auth(username, password):
return username == USERNAME and password == PASSWORD

def authenticate():
return Response('Could not verify your access level for that URL.\n'
'You have to login with proper credentials', 401,
{'WWW-Authenticate': 'Basic realm="Login Required"'})

def requires_auth(f):
@wraps(f)
def decorated(*args, **kwargs):
auth = request.authorization
if not auth or not check_auth(auth.username, auth.password):
return authenticate()
return f(*args, **kwargs)
return decorated

@app.route("/", methods=["GET", "POST"])
def index():
if request.method == "POST":
file = request.files.get("file")
if not file:
flash("No file uploaded.", "error")
return redirect(request.url)

filename = secure_filename(file.filename)
filepath = os.path.join(UPLOAD_FOLDER, filename)
file.save(filepath)

try:
if filename.endswith(".pdf"):
text = extract_text_from_pdf(filepath)
elif filename.endswith(".html") or filename.endswith(".htm"):
text = extract_text_from_html(filepath)
else:
raise ValueError("Unsupported file type. Only PDF and HTML supported.")

config = load_config(CONFIG_PATH)
summary, out_dir = summarize_document(text, config, SUMMARY_FOLDER)
download_links = save_summary_files(summary, out_dir)

return render_template("index.html", summary=summary, links=download_links)

except Exception as e:
with open(LOG_FILE, "a") as log:
log.write(f"\nERROR: {str(e)}\n")
flash("An error occurred: " + str(e), "error")
return redirect(request.url)

return render_template("index.html")
